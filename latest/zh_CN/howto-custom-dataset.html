操作指南：添加自定义数据集#

fl-sim 中的联邦数据集由两部分组成： 原始数据 与把它分散到各模拟客户端的 划分方案 。本指南先走完 test-files/custom_dataset.py 的完整示例（FedProx 论文所用 FEMNIST 变体的复刻，经 test-files/custom_conf.yml 在 CI 中执行），再从 NumPy 数组出发给出最小模板。

选择基类#

  • FedVisionDataset——图像数据，带常规的 TensorDataset 管线；示例用的就是它；

  • FedDataset——完全抽象的基类，适合一切非视觉数据。

必需接口很小：在 _preload 里加载数据，在 get_dataloader 里按客户端分发dataloader，在 evaluate 里为预测打分。

完整示例#

@register_fed_dataset()
class CustomFEMNIST(FedVisionDataset):
    """
    (part of) federeated EMNIST used in the FedProx paper and the FedDR paper.

    `The FEMNIST data we used in the paper is a subsampled (and repartitioned) version
    of the original full dataset in order to impose additional statistical heterogeneity.
    The above dataset is generated by the following instruction: ...`

    NOTE that this dataset is not the same as the original FEMNIST dataset,
    containing only 10 classes (a-j), instead of 62 classes (a-z, A-Z, 0-9).
    The raw data has been processed using min-max normalization to range [0, 1],
    hence any further augmentation is perhaps inappropriate.

    Parameters
    ----------
    datadir : Union[Path, str], optional
        Directory to store data.
        If ``None``, use default directory.
    transform : Union[str, Callable], default "none"
        Transform to apply to data. Conventions:
        ``"none"`` means no transform, using TensorDataset.
    seed : int, default 0
        Random seed for data partitioning.
    **extra_config : dict, optional
        Extra configurations.

    References
    ----------
    .. [1] https://github.com/litian96/FedProx/tree/master/data/nist

    """

    __name__ = "CustomFEMNIST"

    # --- begin: preload ---
    def _preload(self, datadir: Optional[Union[str, Path]] = None) -> None:
        self.datadir = Path(datadir or CUSTOM_FEMNIST_DATA_DIR).expanduser().resolve()

        self.DEFAULT_TRAIN_CLIENTS_NUM = 200
        self.DEFAULT_TEST_CLIENTS_NUM = 200
        self.DEFAULT_BATCH_SIZE = 20
        self.DEFAULT_TRAIN_FILE = Path("train") / Path("mytrain.json")
        self.DEFAULT_TEST_FILE = Path("test") / Path("mytest.json")

        self._EXAMPLE = "user_data"
        self._IMGAE = "x"
        self._LABEL = "y"

        if self.transform != "none":
            warnings.warn(
                "The images are not raw pixels, but processed. " "The transform argument will be ignored.",
                RuntimeWarning,
            )
            self.transform = "none"

        self.criterion = torch.nn.CrossEntropyLoss()

        self.download_if_needed()

        # client id list
        train_file_path = self.datadir / self.DEFAULT_TRAIN_FILE
        test_file_path = self.datadir / self.DEFAULT_TEST_FILE
        self._train_data_dict = json.loads(train_file_path.read_text())
        self._test_data_dict = json.loads(test_file_path.read_text())
        self._client_ids_train = self._train_data_dict["users"]
        self._client_ids_test = self._test_data_dict["users"]

        self._n_class = len(
            np.unique(
                np.concatenate(
                    [
                        self._train_data_dict[self._EXAMPLE][self._client_ids_train[idx]][self._LABEL]
                        for idx in range(self.DEFAULT_TRAIN_CLIENTS_NUM)
                    ]
                )
            )
        )

_preload 在构造时只跑一次：解析数据目录、声明框架会查询的默认值（客户端数、批大小、文件布局）、设置 criterion、按需下载，并把原始文件读入内存。fl-sim 数据集 构造后不可变 ——划分就在这里做，且只做一次。

    def get_dataloader(
        self,
        train_bs: Optional[int] = None,
        test_bs: Optional[int] = None,
        client_idx: Optional[int] = None,
    ) -> Tuple[torchdata.DataLoader, torchdata.DataLoader]:
        # load data
        if client_idx is None:
            # get ids of all clients
            train_ids = self._client_ids_train
            test_ids = self._client_ids_test
        else:
            # get ids of single client
            train_ids = [self._client_ids_train[client_idx]]
            test_ids = [self._client_ids_test[client_idx]]

        # load data
        train_x = np.vstack([self._train_data_dict[self._EXAMPLE][client_id][self._IMGAE] for client_id in train_ids])
        train_y = np.concatenate([self._train_data_dict[self._EXAMPLE][client_id][self._LABEL] for client_id in train_ids])
        test_x = np.vstack([self._test_data_dict[self._EXAMPLE][client_id][self._IMGAE] for client_id in test_ids])
        test_y = np.concatenate([self._test_data_dict[self._EXAMPLE][client_id][self._LABEL] for client_id in test_ids])

        # dataloader
        train_ds = torchdata.TensorDataset(
            torch.from_numpy(train_x.reshape((-1, 28, 28)).astype(np.float32)).unsqueeze(1),
            torch.from_numpy(train_y.astype(np.int64)),
        )
        train_dl = torchdata.DataLoader(
            dataset=train_ds,
            batch_size=train_bs or self.DEFAULT_BATCH_SIZE,
            shuffle=True,
            drop_last=False,
        )

        test_ds = torchdata.TensorDataset(
            torch.from_numpy(test_x.reshape((-1, 28, 28)).astype(np.float32)).unsqueeze(1),
            torch.from_numpy(test_y.astype(np.int64)),
        )
        test_dl = torchdata.DataLoader(
            dataset=test_ds,
            batch_size=test_bs or self.DEFAULT_BATCH_SIZE,
            shuffle=True,
            drop_last=False,
        )

        return train_dl, test_dl

get_dataloader 的契约：给定客户端索引（或 None 表示"全部"），返回一对开启洗牌的 dataloader (train, test) —— 框架每轮都会重新请求它们；在 rng_mode="per_client" （ 教程：并行与可复现的运行 ）下，被按客户端重新播种的正是这里的洗牌。

    def evaluate(self, probs: torch.Tensor, truths: torch.Tensor) -> Dict[str, float]:
        return {
            "acc": top_n_accuracy(probs, truths, 1),
            "top3_acc": top_n_accuracy(probs, truths, 3),
            "top5_acc": top_n_accuracy(probs, truths, 5),
            "loss": self.criterion(probs, truths).item(),
            "num_samples": probs.shape[0],
        }

evaluate 把一批预测与标签变成写进日志的指标字典（ acc 、 loss 、 num_samples 等——正是 快速上手 读回的那些键）。

最小模板#

从 NumPy 数组到可用的联邦数据集。下述模板已通过端到端执行验证；它编码的三个约定容易漏掉、调起来又贵：

  • define your own ``__init__``: the abstract FedDataset base has no constructor of its own (the vision subclass supplies one); accept (datadir, transform, seed, **extra_config) and re-seed via set_seed so every constructor re-seeds the global RNG exactly like the built-ins do;

  • the framework requests global loaders too: get_dataloader must handle client_idx=None ("all clients"), not just per-client indices;

  • ``candidate_models`` should return fl-sim models (they carry the .device attribute the device-allocation step relies on).

两个 load_partition_data* 方法看似必需，实为训练循环从不调用的遗留 tff 风格批量加载 API——桩实现即可满足抽象接口：

import numpy as np
import torch
import torch.utils.data as torchdata

from fl_sim.models import nn as mnn
from fl_sim.utils.misc import set_seed
from fl_sim.data_processing._register import register_fed_dataset
from fl_sim.data_processing.fed_dataset import FedDataset


@register_fed_dataset()
class MyArrays(FedDataset):
    __name__ = "MyArrays"

    def __init__(self, datadir=None, transform="none", seed=0, **extra_config):
        super().__init__()
        self.datadir = datadir
        self.transform = transform
        self.seed = seed
        set_seed(seed)  # so that every constructor re-seeds, as the built-ins do
        for k, v in extra_config.items():
            setattr(self, k, v)
        self._preload()

    def _preload(self, datadir=None):
        rng = np.random.default_rng(self.seed)
        n_clients = 30
        self._parts = [  # one (features, labels) pair per client
            (
                rng.normal(size=(100, 20)).astype(np.float32),
                rng.integers(0, 2, size=100).astype(np.int64),
            )
            for _ in range(n_clients)
        ]
        self.DEFAULT_TRAIN_CLIENTS_NUM = n_clients
        self.DEFAULT_TEST_CLIENTS_NUM = n_clients
        self.DEFAULT_BATCH_SIZE = 32
        self.criterion = torch.nn.CrossEntropyLoss()

    def get_dataloader(self, train_bs=None, test_bs=None, client_idx=None):
        parts = self._parts if client_idx is None else [self._parts[client_idx]]
        x = np.concatenate([p[0] for p in parts])
        y = np.concatenate([p[1] for p in parts])
        ds = torchdata.TensorDataset(torch.from_numpy(x), torch.from_numpy(y))
        dl = torchdata.DataLoader(ds, batch_size=train_bs or self.DEFAULT_BATCH_SIZE, shuffle=True)
        return dl, dl  # train and test loaders; a real dataset separates them

    def evaluate(self, probs, truths):
        return {
            "acc": (probs.argmax(1) == truths).float().mean().item(),
            "loss": self.criterion(probs, truths).item(),
            "num_samples": probs.shape[0],
        }

    # legacy tff-style bulk loaders; the training loop never calls them
    def load_partition_data(self, batch_size=None):
        raise NotImplementedError

    def load_partition_data_distributed(self, process_id, batch_size=None):
        raise NotImplementedError

    @property
    def url(self):
        return ""  # nothing to download

    @property
    def candidate_models(self):
        # fl-sim models carry a .device attribute the framework relies on
        return {"mlp": mnn.MLP(dim_in=20, dim_out=2, ndim=0)}

    @property
    def doi(self):
        return []

使用#

应用 @register_fed_dataset() 装饰器后，只要模块被导入过，数据集即可按名字构造（或以模块 路径 作为数据集 name ——见 FedDataArgs），且所有内置组件原样可用：

import my_arrays  # the module above; registers the dataset

from fl_sim.data_processing import FedDataArgs
from fl_sim.algorithms.fedopt import FedAvgClientConfig, FedAvgServer, FedAvgServerConfig

dataset = FedDataArgs(name="MyArrays")._create_fed_dataset()
server = FedAvgServer(
    dataset.candidate_models["mlp"],
    dataset,
    FedAvgServerConfig(num_iters=5, num_clients=10, clients_sample_ratio=1.0),
    FedAvgClientConfig(num_epochs=1, batch_size=32),
)
server.train()  # identical to :doc:`quickstart` from here on
